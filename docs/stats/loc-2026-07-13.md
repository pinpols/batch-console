## 复算

```bash
# 按类型
for ext in vue ts css json md sh; do
  f=$(find . -type d \( -name node_modules -o -name dist -o -name .git -o -name coverage -o -name playwright-report -o -name test-results \) -prune -o -name "*.$ext" -type f -print 2>/dev/null | wc -l)
  l=$(find . -type d \( -name node_modules -o -name dist -o -name .git -o -name coverage -o -name playwright-report -o -name test-results \) -prune -o -name "*.$ext" -type f -print 2>/dev/null | xargs cat 2>/dev/null | wc -l)
  printf ".%-5s %5s files %8s lines\n" "$ext" "$f" "$l"
done

# src 主代码口径
find src -type f \( -name '*.vue' -o -name '*.ts' -o -name '*.css' \) | xargs cat | wc -l   # src 全量
wc -l src/types/api.generated.ts                                                            # 生成
find src -name '*.spec.ts' -o -name '*.test.ts' | xargs cat | wc -l                        # src 内单测

# 测试
find e2e -name '*.ts' -type f | xargs cat | wc -l
```
